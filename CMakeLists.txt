# C++20 thermal analogy by Jason Stys. Targets: reactor_model, reactor_core, core_tests.
# Variables/options: REACTOR_SANITIZE, REACTOR_COVERAGE. Exact lines: docs/CODE_INDEX.md.
cmake_minimum_required(VERSION 3.28)
project(ReactorSystemsSimulator VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(REACTOR_SANITIZE "Enable Address/UndefinedBehavior sanitizer checks" OFF)
option(REACTOR_COVERAGE "Enable GNU coverage instrumentation" OFF)
if(REACTOR_SANITIZE AND MSVC)
  message(FATAL_ERROR "Combined ASan/UBSan mode requires the documented Clang Linux toolchain")
endif()
if(REACTOR_COVERAGE AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Coverage mode requires GCC; refusing an uninstrumented coverage build")
endif()
add_library(reactor_model native/model.cpp)
target_include_directories(reactor_model PUBLIC native)
add_executable(reactor_core native/cli.cpp)
target_link_libraries(reactor_core PRIVATE reactor_model)
add_executable(core_tests native/tests.cpp)
target_link_libraries(core_tests PRIVATE reactor_model)
foreach(target reactor_model reactor_core core_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(REACTOR_SANITIZE AND NOT MSVC)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(REACTOR_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endforeach()
enable_testing()
add_test(NAME numerical_contracts COMMAND core_tests)
add_test(NAME cli_baseline COMMAND reactor_core baseline 300 1 1 1 0 0.1)
add_test(NAME cli_rejects_invalid COMMAND reactor_core baseline 300 nan 1 1 0 0.1)
set_tests_properties(cli_rejects_invalid PROPERTIES WILL_FAIL TRUE)
